# GlusterFS Cluster Configuration

# Cluster Settings
GLUSTER_CLUSTER_NAME=my-gluster-cluster
GLUSTER_NODE_IPS=172.20.0.10,172.20.0.11,172.20.0.12
GLUSTER_REPLICA_COUNT=3
GLUSTER_NETWORK_SUBNET=172.20.0.0/16
GLUSTER_NETWORK_NAME=gluster-net

# Docker Images (use your own registry)
GLUSTER_NODE_IMAGE=your-registry.com/gluster-cluster/node:latest
GLUSTER_MANAGER_IMAGE=your-registry.com/gluster-cluster/manager:latest
GLUSTER_CLIENT_IMAGE=your-registry.com/gluster-cluster/client:latest

# Storage Configuration
GLUSTER_DATA_PATH=/data/glusterfs
GLUSTER_BACKUP_PATH=./backups

# Volume Configuration
# Format: name:type:replica_count:host_paths:mount_point;name2:type2:...
GLUSTER_VOLUMES=shared:replicated:3:/mnt/shared/tomcat-resources,/mnt/shared/app-data:/mnt/shared;data:replicated:3:/mnt/data:/mnt/data

# Performance Settings
GLUSTER_CACHE_SIZE=256MB
GLUSTER_WRITE_BUFFER_SIZE=4MB
GLUSTER_ALLOW_INSECURE=true

# Logging
GLUSTER_LOG_LEVEL=info
GLUSTER_LOG_PATH=/var/log/glusterfs

# Port Configuration
GLUSTER_NODE1_PORT=24007
GLUSTER_NODE1_BRICK_PORT=49152
GLUSTER_NODE2_PORT=24008
GLUSTER_NODE2_BRICK_PORT=49153
GLUSTER_NODE3_PORT=24009
GLUSTER_NODE3_BRICK_PORT=49154

# Application Configuration
TOMCAT_PORT=8080
